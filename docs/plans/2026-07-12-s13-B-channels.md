```python
from dataclasses import dataclass

from app.core.exceptions import ResourceNotFound, ValidationFailed
from app.modules.map import services as map_services
from app.modules.spots import services as spots_services

CHANNEL_LABELS = {"around": "Around", "hot": "Hot", "hidden": "Hidden",
                  "festa": "Festa", "pets": "Pets", "snap": "Snap"}
_CARD_COUNT = 10


@dataclass(frozen=True)
class ChannelCardRow:
    content_id: str | None
    title: str
    region_label: str
    image_url: str | None
    dist: float | None = None
    rank: int | None = None
    dday: str | None = None
    line: str | None = None
    tag: str | None = None
    saveable: bool = True


async def load_channel_cards(session, redis, kto, *, key: str,
                             lat: float | None, lng: float | None) -> list[ChannelCardRow]:
    if key not in CHANNEL_LABELS:
        raise ResourceNotFound(f"unknown channel {key}")
    if key == "around":
        if lat is None or lng is None:
            raise ValidationFailed("around channel requires lat/lng")
        cards = await map_services.nearby_cards(
            session, lat=lat, lng=lng, radius=5000, category=None,
            sw_lat=None, sw_lng=None, ne_lat=None, ne_lng=None)
        return [ChannelCardRow(content_id=c.contentId, title=c.title,
                               region_label=_region(c), image_url=c.firstImageUrl,
                               dist=c.dist) for c in cards[:_CARD_COUNT]]
    if key == "hot":
        rows = await spots_services.load_hot_spots(session, limit=_CARD_COUNT)
        return [ChannelCardRow(content_id=r.content_id, title=r.title,
                               region_label=r.region_label, image_url=r.first_image_url,
                               rank=r.rank) for r in rows]
    if key == "hidden":
        rows = await spots_services.load_hidden_spots(session, limit=_CARD_COUNT)
        return [ChannelCardRow(content_id=r.content_id, title=r.title,
                               region_label=r.region_label, image_url=r.first_image_url)
                for r in rows]
    return await _load_kto_channel(redis, kto, key)
```

```python
KST = timezone(timedelta(hours=9))

def _today() -> date:

def _ttl_until_kst_midnight() -> int:
    now = datetime.now(KST)
    midnight = (now + timedelta(days=1)).replace(hour=0, minute=0, second=0, microsecond=0)
    return max(int((midnight - now).total_seconds()), 60)


async def fetch_festa_cards(kto: KtoClient, *, today: date | None = None) -> list[ChannelCardRow]:
    today = today or _today()
    items = await kto.call(KtoService.KOR, "searchFestival2",
                           eventStartDate=today.strftime("%Y%m%d"), numOfRows=_ROWS, arrange="C")
    cards = []
    for it in items:
        end = _parse_ymd(it.get("eventenddate"))
        start = _parse_ymd(it.get("eventstartdate"))
        img = it.get("firstimage") or None
        if not img or (end and end < today):
            continue
        ongoing = start is not None and start <= today
        target = end if ongoing else start
        dday = f"D-{max((target - today).days, 0)}" if target else None
        line = _festa_line(start, end, it.get("addr1"), ongoing)
        cards.append(ChannelCardRow(content_id=str(it["contentid"]), title=it["title"],
                                    region_label=_short_addr(it.get("addr1")), image_url=img,
                                    dday=dday, line=line))
    cards.sort(key=lambda c: c.dday or "D-999")
    return cards[:10]
```
`_festa_line`: 진행 중 → "7월 15일까지 · 무주 예체문화관 일원", 임박 → "7월 19일부터 · …".

```python
async def load_kto_channel_cached(redis, kto, key: str) -> list[ChannelCardRow]:
    cache_key = f"channel:{key}:v1"

    except Exception:
        pass
```
